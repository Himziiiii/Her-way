import { useEffect, useMemo, useState } from 'react';
import {
  ThumbsUp,
  Flag,
  MapPin,
  Clock,
  ShieldCheck,
  AlertTriangle,
  ShieldAlert,
} from 'lucide-react';
import { REPORT_CATEGORIES, voteOnReport } from '../../services/communityService';

function formatRelativeTime(isoString, nowMs) {
  if (!isoString) return 'Just now';
  const diffMs = nowMs - new Date(isoString).getTime();
  const mins = Math.max(1, Math.floor(diffMs / 60000));
  if (mins < 60) return `${mins}m ago`;
  const hrs = Math.floor(mins / 60);
  if (hrs < 24) return `${hrs}h ago`;
  const days = Math.floor(hrs / 24);
  return `${days}d ago`;
}

function formatDistanceMeters(meters) {
  if (typeof meters !== 'number') return null;
  if (meters < 1000) return `${Math.round(meters)} m away`;
  return `${(meters / 1000).toFixed(1)} km away`;
}

export default function DailyUpdatesTab({
  reports = [],
  isLoading = false,
  onFocusLocation,
  onReportVoted,
  onOpenCreate,
}) {
  const [sentimentFilter, setSentimentFilter] = useState('all');
  const [maxHoursFilter, setMaxHoursFilter] = useState(48);
  const [busyVoteId, setBusyVoteId] = useState(null);
  const [voteFeedback, setVoteFeedback] = useState('');
  const [nowMs, setNowMs] = useState(() => Date.now());

  useEffect(() => {
    const id = setInterval(() => setNowMs(Date.now()), 60000);
    return () => clearInterval(id);
  }, []);

  const categoryMap = useMemo(() => {
    const map = {};
    for (const c of REPORT_CATEGORIES) {
      map[c.value] = c;
    }
    return map;
  }, []);

  const filteredReports = useMemo(() => {
    const cutoffMs = nowMs - maxHoursFilter * 60 * 60 * 1000;

    return reports.filter((rep) => {
      const createdMs = rep.created_at ? new Date(rep.created_at).getTime() : nowMs;
      if (createdMs < cutoffMs) return false;

      if (sentimentFilter === 'positive') {
        return rep.sentiment === 'positive';
      }
      if (sentimentFilter === 'alerts') {
        return rep.sentiment === 'caution' || rep.sentiment === 'danger';
      }
      return true;
    });
  }, [reports, sentimentFilter, maxHoursFilter, nowMs]);

  async function handleVote(reportId, voteType) {
    setBusyVoteId(reportId);
    setVoteFeedback('');
    try {
      await voteOnReport(reportId, voteType);
      onReportVoted?.();
    } catch (err) {
      setVoteFeedback(err?.message || 'Sign in to vote on updates.');
    } finally {
      setBusyVoteId(null);
    }
  }

  return (
    <div className="space-y-3">
      <div className="flex flex-wrap items-center justify-between gap-2">
        <div className="flex items-center gap-1">
          {[
            { id: 'all', label: 'All' },
            { id: 'positive', label: 'Safe Spots' },
            { id: 'alerts', label: 'Alerts' },
          ].map((pill) => (
            <button
              key={pill.id}
              type="button"
              onClick={() => setSentimentFilter(pill.id)}
              className={`rounded-full px-2.5 py-1 text-[11px] font-semibold transition ${
                sentimentFilter === pill.id
                  ? 'bg-slate-900 text-white'
                  : 'bg-slate-100 text-slate-600 hover:bg-slate-200'
              }`}
            >
              {pill.label}
            </button>
          ))}
        </div>

        <select
          aria-label="Filter updates by age"
          value={maxHoursFilter}
          onChange={(e) => setMaxHoursFilter(Number(e.target.value))}
          className="rounded-lg border border-slate-200 bg-white px-2 py-1 text-[11px] font-medium text-slate-700 focus:border-rose-500 focus:outline-none"
        >
          <option value={6}>Last 6 hrs</option>
          <option value={24}>Last 24 hrs</option>
          <option value={48}>Last 48 hrs</option>
        </select>
      </div>

      {voteFeedback && (
        <p className="rounded-xl bg-amber-50 px-3 py-1.5 text-xs text-amber-800">
          {voteFeedback}
        </p>
      )}

      {isLoading ? (
        <div className="space-y-2">
          {[1, 2, 3].map((n) => (
            <div key={n} className="h-20 animate-pulse rounded-xl bg-slate-100" />
          ))}
        </div>
      ) : filteredReports.length === 0 ? (
        <div className="rounded-xl border border-dashed border-slate-200 p-5 text-center">
          <p className="text-xs font-semibold text-slate-700">
            No updates match this filter nearby yet
          </p>
          <p className="mt-1 text-[11px] text-slate-500">
            Share street lighting, crowd levels, or patrol updates to help others walking nearby.
          </p>
          <button
            type="button"
            onClick={onOpenCreate}
            className="mt-3 rounded-xl bg-rose-50 px-3 py-1.5 text-xs font-semibold text-rose-700 hover:bg-rose-100"
          >
            + Add First Update
          </button>
        </div>
      ) : (
        <ul className="max-h-[380px] space-y-2.5 overflow-y-auto pr-0.5">
          {filteredReports.map((rep) => {
            const meta = categoryMap[rep.category];
            const isPositive = rep.sentiment === 'positive';
            const isDanger = rep.sentiment === 'danger';

            const badgeClasses = isPositive
              ? 'bg-emerald-50 text-emerald-700 border-emerald-200'
              : isDanger
                ? 'bg-rose-50 text-rose-700 border-rose-200'
                : 'bg-amber-50 text-amber-800 border-amber-200';

            const StatusIcon = isPositive
              ? ShieldCheck
              : isDanger
                ? ShieldAlert
                : AlertTriangle;

            const distText = formatDistanceMeters(rep.distance_meters);

            return (
              <li
                key={rep.id}
                className="rounded-xl border border-slate-200/90 bg-slate-50/40 p-3 transition hover:border-slate-300 hover:bg-white"
              >
                <div className="flex items-start justify-between gap-2">
                  <span
                    className={`inline-flex items-center gap-1 rounded-full border px-2.5 py-0.5 text-[11px] font-semibold ${badgeClasses}`}
                  >
                    <StatusIcon className="h-3 w-3" />
                    <span>{meta?.label || rep.category.replace(/_/g, ' ')}</span>
                  </span>

                  <span className="inline-flex items-center gap-1 text-[11px] text-slate-400">
                    <Clock className="h-3 w-3" />
                    {formatRelativeTime(rep.created_at, nowMs)}
                  </span>
                </div>

                {rep.area_name && (
                  <button
                    type="button"
                    onClick={() =>
                      onFocusLocation?.({
                        lat: rep.lat,
                        lng: rep.lng,
                        label: rep.area_name,
                      })
                    }
                    className="mt-1.5 inline-flex items-center gap-1 text-left text-xs font-semibold text-slate-800 hover:text-rose-600"
                  >
                    <MapPin className="h-3.5 w-3.5 shrink-0 text-rose-500" />
                    <span className="line-clamp-1">{rep.area_name}</span>
                    {distText && (
                      <span className="shrink-0 font-normal text-slate-400">({distText})</span>
                    )}
                  </button>
                )}

                {rep.note && (
                  <p className="mt-1 text-xs leading-relaxed text-slate-600">{rep.note}</p>
                )}

                <div className="mt-2.5 flex items-center justify-between border-t border-slate-100 pt-2">
                  <button
                    type="button"
                    disabled={busyVoteId === rep.id}
                    onClick={() => handleVote(rep.id, 'upvote')}
                    className="inline-flex items-center gap-1 rounded-lg px-2 py-1 text-[11px] font-semibold text-slate-600 transition hover:bg-emerald-50 hover:text-emerald-700 disabled:opacity-50"
                  >
                    <ThumbsUp className="h-3 w-3" />
                    <span>Helpful ({rep.upvotes || 0})</span>
                  </button>

                  <button
                    type="button"
                    disabled={busyVoteId === rep.id}
                    onClick={() => handleVote(rep.id, 'flag')}
                    title="Flag inaccurate or outdated report"
                    className="inline-flex items-center gap-1 rounded-lg px-2 py-1 text-[11px] font-medium text-slate-400 transition hover:bg-rose-50 hover:text-rose-600 disabled:opacity-50"
                  >
                    <Flag className="h-3 w-3" />
                    <span>Flag</span>
                  </button>
                </div>
              </li>
            );
          })}
        </ul>
      )}
    </div>
  );
}

import { useEffect, useState } from 'react';
import {
  X,
  MapPin,
  Navigation,
  Send,
  Loader2,
  Lightbulb,
  AlertCircle,
  Compass,
} from 'lucide-react';
import {
  REPORT_CATEGORIES,
  createRouteReport,
  createRouteSuggestion,
} from '../../services/communityService';
import { reverseGeocode } from '../../services/geoProvider';

function QuickReportModalBody({
  onClose,
  initialMode,
  initialCoords,
  origin,
  destination,
  onSubmitted,
}) {
  const [mode, setMode] = useState(initialMode);
  const [coords, setCoords] = useState(initialCoords);
  const [areaName, setAreaName] = useState(initialCoords?.label || '');
  const [category, setCategory] = useState('well_lit');
  const [note, setNote] = useState('');

  const [originArea, setOriginArea] = useState(origin?.label || '');
  const [destinationArea, setDestinationArea] = useState(destination?.label || '');
  const [title, setTitle] = useState('');
  const [advice, setAdvice] = useState('');
  const [timeWindow, setTimeWindow] = useState('evening_night');

  const [isLocating, setIsLocating] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [errorMsg, setErrorMsg] = useState('');

  useEffect(() => {
    if (!initialCoords?.lat || !initialCoords?.lng || initialCoords?.label) return undefined;
    let active = true;
    reverseGeocode(initialCoords.lat, initialCoords.lng).then((lbl) => {
      if (active) setAreaName(lbl);
    });
    return () => {
      active = false;
    };
  }, [initialCoords?.lat, initialCoords?.lng, initialCoords?.label]);

  function handleGrabCurrentGPS() {
    if (!navigator.geolocation) {
      setErrorMsg('Browser geolocation is unavailable.');
      return;
    }
    setIsLocating(true);
    setErrorMsg('');
    navigator.geolocation.getCurrentPosition(
      async (pos) => {
        const lat = pos.coords.latitude;
        const lng = pos.coords.longitude;
        setCoords({ lat, lng });
        const resolved = await reverseGeocode(lat, lng);
        setAreaName(resolved);
        setIsLocating(false);
      },
      () => {
        setIsLocating(false);
        setErrorMsg('Could not access GPS. Tap a spot on the map first or allow location.');
      },
      { enableHighAccuracy: true, timeout: 10000 }
    );
  }

  async function handleSubmit(e) {
    e.preventDefault();
    setErrorMsg('');
    setIsSubmitting(true);

    try {
      if (mode === 'report') {
        if (!coords?.lat || !coords?.lng) {
          throw new Error('Select a location via GPS or tap on the map first.');
        }

        const created = await createRouteReport({
          lat: coords.lat,
          lng: coords.lng,
          areaName: areaName || `${coords.lat.toFixed(4)}, ${coords.lng.toFixed(4)}`,
          category,
          note,
        });

        onSubmitted?.({ type: 'report', item: created });
        onClose?.();
      } else {
        const startLat = origin?.lat ?? coords?.lat;
        const startLng = origin?.lng ?? coords?.lng;
        const endLat = destination?.lat ?? startLat;
        const endLng = destination?.lng ?? startLng;

        if (!startLat || !startLng) {
          throw new Error('Set at least a start location or GPS point for this route tip.');
        }
        if (!originArea.trim() || !destinationArea.trim()) {
          throw new Error('Enter both From and To area names.');
        }
        if (title.trim().length < 3 || advice.trim().length < 5) {
          throw new Error('Add a short headline and helpful route advice.');
        }

        const created = await createRouteSuggestion({
          originArea,
          destinationArea,
          startLat,
          startLng,
          endLat,
          endLng,
          title,
          advice,
          timeWindow,
        });

        onSubmitted?.({ type: 'suggestion', item: created });
        onClose?.();
      }
    } catch (err) {
      setErrorMsg(err?.message || 'Unable to submit update.');
    } finally {
      setIsSubmitting(false);
    }
  }

  return (
    <div
      className="fixed inset-0 z-[1000] flex items-end justify-center bg-slate-900/50 p-3 backdrop-blur-xs sm:items-center"
      role="dialog"
      aria-modal="true"
      aria-labelledby="quick-report-title"
    >
      <div className="w-full max-w-lg overflow-hidden rounded-2xl border border-slate-200 bg-white shadow-xl">
        <div className="flex items-center justify-between border-b border-slate-100 px-5 py-3.5">
          <div className="flex items-center gap-2">
            <span className="flex h-8 w-8 items-center justify-center rounded-xl bg-rose-50 text-rose-600">
              {mode === 'report' ? <Lightbulb className="h-4 w-4" /> : <Compass className="h-4 w-4" />}
            </span>
            <div>
              <h2 id="quick-report-title" className="text-sm font-bold text-slate-900">
                Community Safety Contribution
              </h2>
              <p className="text-xs text-slate-500">Helps score routes for everyone nearby</p>
            </div>
          </div>

          <button
            type="button"
            onClick={onClose}
            aria-label="Close modal"
            className="rounded-lg p-1.5 text-slate-400 hover:bg-slate-100 hover:text-slate-700"
          >
            <X className="h-4 w-4" />
          </button>
        </div>

        <div className="grid grid-cols-2 gap-1 bg-slate-50 p-1.5">
          <button
            type="button"
            onClick={() => {
              setMode('report');
              setErrorMsg('');
            }}
            className={`rounded-xl py-2 text-xs font-semibold transition ${
              mode === 'report'
                ? 'bg-white text-rose-600 shadow-xs'
                : 'text-slate-600 hover:text-slate-900'
            }`}
          >
            Daily Route Update
          </button>
          <button
            type="button"
            onClick={() => {
              setMode('suggestion');
              setErrorMsg('');
            }}
            className={`rounded-xl py-2 text-xs font-semibold transition ${
              mode === 'suggestion'
                ? 'bg-white text-rose-600 shadow-xs'
                : 'text-slate-600 hover:text-slate-900'
            }`}
          >
            Safer Route Suggestion
          </button>
        </div>

        <form onSubmit={handleSubmit} className="max-h-[78vh] space-y-4 overflow-y-auto p-5">
          {mode === 'report' ? (
            <>
              <div className="rounded-xl border border-slate-200 bg-slate-50/70 p-3">
                <div className="flex items-center justify-between gap-2">
                  <span className="inline-flex items-center gap-1.5 text-xs font-semibold text-slate-700">
                    <MapPin className="h-3.5 w-3.5 text-rose-600" />
                    Pinned Spot
                  </span>
                  <button
                    type="button"
                    onClick={handleGrabCurrentGPS}
                    disabled={isLocating}
                    className="inline-flex items-center gap-1 text-xs font-semibold text-rose-600 hover:text-rose-700"
                  >
                    {isLocating ? (
                      <Loader2 className="h-3.5 w-3.5 animate-spin" />
                    ) : (
                      <Navigation className="h-3.5 w-3.5" />
                    )}
                    <span>Use Current GPS</span>
                  </button>

                <input

                  placeholder="Street, gate, metro exit, or landmark name"
                  className="mt-2 w-full rounded-lg border border-slate-200 bg-white px-3 py-2 text-xs text-slate-900 placeholder:text-slate-400 focus:border-rose-500 focus:outline-none"
                />

                {coords?.lat && coords?.lng ? (
                  <p className="mt-1.5 text-[11px] text-slate-500">
                    Coordinates: {coords.lat.toFixed(5)}, {coords.lng.toFixed(5)}
                  </p>
                ) : (
                  <p className="mt-1.5 text-[11px] text-amber-700">
                    No pin set yetâ€”click &ldquo;Use Current GPS&rdquo; or tap directly on the map.
                  </p>
                )}
              </div>

                <div className="grid grid-cols-1 gap-2 sm:grid-cols-2">

function MapViewportController({ origin, destination, activeRoute, livePosition, followLive }) {
  const map = useMap();

  useEffect(() => {
    if (followLive && livePosition?.lat && livePosition?.lng) {
      map.setView([livePosition.lat, livePosition.lng], Math.max(map.getZoom(), 15), {
        animate: true,
      });
      return;
    }

    if (activeRoute?.latLngCoords?.length > 1) {
      const bounds = L.latLngBounds(activeRoute.latLngCoords);
      map.fitBounds(bounds, { padding: [48, 48], maxZoom: 16 });
      return;
    }

    if (origin?.lat && origin?.lng && destination?.lat && destination?.lng) {
      const bounds = L.latLngBounds([
        [origin.lat, origin.lng],
        [destination.lat, destination.lng],
      ]);
      map.fitBounds(bounds, { padding: [56, 56], maxZoom: 15 });
      return;
    }

    if (origin?.lat && origin?.lng) {
      map.setView([origin.lat, origin.lng], 14, { animate: true });
    }
  }, [map, origin, destination, activeRoute, livePosition, followLive]);

  return null;
}

import { useMemo } from 'react';
import { Polyline } from 'react-leaflet';

export default function RouteOverlay({
  routes = [],
  selectedRouteId = null,
  onSelectRoute,
}) {
  const activeRoute = useMemo(
    () => routes.find((r) => r.id === selectedRouteId) || routes[0] || null,
    [routes, selectedRouteId]
  );

  const otherRoutes = useMemo(
    () => routes.filter((r) => r.id !== activeRoute?.id && Array.isArray(r.latLngCoords)),
    [routes, activeRoute]
  );

  return (
    <>
      {/* Unselected alternative routes */}
      {otherRoutes.map((route) => (
        <Polyline
          key={route.id}
          positions={route.latLngCoords}
          pathOptions={{
            color: '#64748b',
            weight: 5,
            opacity: 0.45,
            dashArray: '6 8',
          }}
          eventHandlers={{
            click: () => onSelectRoute?.(route.id),
          }}
        />
      ))}

      {/* Active route base shadow/outline */}
      {activeRoute?.latLngCoords?.length > 1 && (
        <Polyline
          positions={activeRoute.latLngCoords}
          pathOptions={{
            color: '#0f172a',
            weight: 9,
            opacity: 0.16,
          }}
        />
      )}

      {/* Active route safety-colored segments */}
      {activeRoute?.segments?.length > 0
        ? activeRoute.segments.map((seg, idx) => (
            <Polyline
              key={`${activeRoute.id}-seg-${idx}`}
              positions={seg.positions}
              pathOptions={{
                color: seg.color,
                weight: 6.5,
                opacity: 0.92,
                lineCap: 'round',
                lineJoin: 'round',
              }}
            />
          ))
        : activeRoute?.latLngCoords?.length > 1 && (
            <Polyline

                color: '#16a34a',

  const mins = Math.max(1, Math.round(seconds / 60));
  if (mins < 60) return `${mins} min`;
  const hrs = Math.floor(mins / 60);
  const rem = mins % 60;
  return rem > 0 ? `${hrs} hr ${rem} min` : `${hrs} hr`;
}

function formatDistance(meters = 0) {
  if (meters < 1000) return `${Math.round(meters)} m`;

// src/context/AuthContext.jsx
import { useCallback, useEffect, useState } from 'react';
import { supabase } from '../services/supabaseClient';
import { AuthContext } from './useAuth';

const DEFAULT_SAFETY_PREFS = {
  prioritize_lighting: true,
  avoid_deserted: true,
  night_mode_strict: true,
  checkin_interval_mins: 15,
};

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [profile, setProfile] = useState(null);
  const [loading, setLoading] = useState(() => Boolean(supabase));

  const loadUserProfile = useCallback(async (authUser) => {
    if (!supabase || !authUser) {
      setProfile(null);
      return null;
    }

    const { data, error } = await supabase
      .from('profiles')
      .select('id, display_name, emergency_contacts, safety_preferences')
      .eq('id', authUser.id)
      .maybeSingle();

    if (error) {
      console.error('Failed to load profile:', error.message);
      return null;
    }

    if (data) {
      const normalized = {
        ...data,
        emergency_contacts: Array.isArray(data.emergency_contacts) ? data.emergency_contacts : [],
        safety_preferences: {
          ...DEFAULT_SAFETY_PREFS,
          ...(data.safety_preferences || {}),
        },
      };
      setProfile(normalized);
      return normalized;
    }

    const fallbackName =
      authUser.user_metadata?.display_name ||
      (authUser.email ? authUser.email.split('@')[0] : 'HerWay Traveler');

    const { data: created } = await supabase
      .from('profiles')
      .upsert({
        id: authUser.id,
        display_name: fallbackName,
        emergency_contacts: [],
        safety_preferences: DEFAULT_SAFETY_PREFS,
      })
      .select('id, display_name, emergency_contacts, safety_preferences')
      .single();

    if (created) {
      setProfile(created);
      return created;
    }
    return null;
  }, []);

  useEffect(() => {
    if (!supabase) return undefined;

    let mounted = true;

    supabase.auth.getSession().then(async ({ data: { session } }) => {
      if (!mounted) return;
      const currentUser = session?.user ?? null;
      setUser(currentUser);
      if (currentUser) {
        await loadUserProfile(currentUser);
      }
      if (mounted) setLoading(false);
    });

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange(async (_event, session) => {
      const nextUser = session?.user ?? null;
      setUser(nextUser);
      if (nextUser) {
        await loadUserProfile(nextUser);
      } else {
        setProfile(null);
      }
      if (mounted) setLoading(false);
    });

    return () => {
      mounted = false;
      subscription.unsubscribe();
    };
  }, [loadUserProfile]);

  async function signInWithEmail(email, password) {
    if (!supabase) throw new Error('Supabase is not configured.');
    const { data, error } = await supabase.auth.signInWithPassword({
      email: email.trim(),
      password,
    });
    if (error) throw new Error(error.message);
    return data;
  }

  async function signUpWithEmail({ email, password, displayName }) {
    if (!supabase) throw new Error('Supabase is not configured.');
    const { data, error } = await supabase.auth.signUp({
      email: email.trim(),
      password,
      options: {
        data: {
          display_name: (displayName || '').trim() || email.split('@')[0],
        },
      },
    });
    if (error) throw new Error(error.message);
    return data;
  }

  async function signOut() {
    if (!supabase) return;
    await supabase.auth.signOut();
    setUser(null);
    setProfile(null);
  }

  async function updateSafetyProfile({ displayName, emergencyContacts, safetyPreferences }) {
    if (!supabase || !user) {
      throw new Error('Sign in to save your safety preferences.');
    }

    const payload = {
      id: user.id,
      display_name:
        typeof displayName === 'string' ? displayName.trim() : profile?.display_name || '',
      emergency_contacts: Array.isArray(emergencyContacts)
        ? emergencyContacts
        : profile?.emergency_contacts || [],
      safety_preferences: {
        ...DEFAULT_SAFETY_PREFS,
        ...(profile?.safety_preferences || {}),
        ...(safetyPreferences || {}),
      },
      updated_at: new Date().toISOString(),
    };

    const { data, error } = await supabase
      .from('profiles')
      .upsert(payload)
      .select('id, display_name, emergency_contacts, safety_preferences')
      .single();

    if (error) {
      throw new Error(error.message || 'Could not update profile.');
    }

    setProfile(data);
    return data;
  }

  const value = {
    user,
    profile,
    loading,
    safetyPreferences: profile?.safety_preferences || DEFAULT_SAFETY_PREFS,
    emergencyContacts: profile?.emergency_contacts || [],
    signInWithEmail,
    signUpWithEmail,
    signOut,
    updateSafetyProfile,
  };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

function formatTimeAgo(isoString, nowMs) {
  if (!isoString) return 'Just now';
  const diffSec = Math.max(0, Math.floor((nowMs - new Date(isoString).getTime()) / 1000));
  if (diffSec < 15) return 'Updated just now';
  if (diffSec < 60) return `${diffSec}s ago`;
  const mins = Math.floor(diffSec / 60);
  if (mins < 60) return `${mins}m ago`;
  return `${Math.floor(mins / 60)}h ago`;
}

    .from('report_votes')
    .select('vote_type')
    .eq('user_id', user.id)
    .eq('report_id', reportId)
    .maybeSingle();

  if (existing?.vote_type === voteType) {
    // Remove vote if clicked again
    const { error } = await client

      .eq('user_id', user.id)

    if (error) throw new Error(error.message);

  const { error } = await client.from('report_votes').upsert(
    {
      user_id: user.id,
      report_id: reportId,
      vote_type: voteType,
    },
    { onConflict: 'user_id,report_id' }
  );

  if (error) throw new Error(error.message);
  return { action: 'saved', voteType };
}

const CACHE_TTL_MS = 20 * 60 * 1000; // 20 minutes
const poiMemoryCache = new Map();

function getStorageCache(key) {
  if (poiMemoryCache.has(key)) {
    const entry = poiMemoryCache.get(key);
    if (Date.now() - entry.timestamp < CACHE_TTL_MS) return entry.data;
    poiMemoryCache.delete(key);
  }

  try {
    const raw = sessionStorage.getItem(key);
    if (!raw) return null;
    const parsed = JSON.parse(raw);
    if (Date.now() - parsed.timestamp < CACHE_TTL_MS) {
      poiMemoryCache.set(key, parsed);
      return parsed.data;
    }
    sessionStorage.removeItem(key);
  } catch {
    // Ignore storage quota or private-mode errors
  }
  return null;
}

function setStorageCache(key, data) {
  const payload = { timestamp: Date.now(), data };
  poiMemoryCache.set(key, payload);
  try {
    sessionStorage.setItem(key, JSON.stringify(payload));
  } catch {
    // Ignore storage errors
  }
}

/**
 * Computes realistic travel time in minutes for 'walk' (~4.8 km/h) vs 'car' (~26 km/h).
 */
export function calculateModeEtaMinutes(distanceMeters, mode = 'walk') {
  const km = Math.max(0.05, (Number(distanceMeters) || 0) / 1000);
  if (mode === 'driving' || mode === 'car') {
    const carSpeedKmh = 26.0;
    return Math.max(2, Math.round((km / carSpeedKmh) * 60 + 2));
  }
  const walkSpeedKmh = 4.8;
  return Math.max(2, Math.round((km / walkSpeedKmh) * 60 + 1));
}

export function formatEtaLabel(minutes) {
  const mins = Math.max(1, Math.round(Number(minutes) || 1));
  if (mins < 60) return `${mins} min`;
  const h = Math.floor(mins / 60);
  const rem = mins % 60;
  return rem === 0 ? `${h} hr` : `${h} hr ${rem} min`;
}

/**
 * Autocomplete place search using Photon with Nominatim fallback.
 * @param {string} query
 * @param {{ lat?: number, lng?: number, limit?: number, signal?: AbortSignal }} options
 */
export async function searchPlaces(query, options = {}) {
  const trimmed = (query || '').trim();
  if (trimmed.length < 2) return [];

  const { lat, lng, limit = 5, signal } = options;

  // 1. Try Photon first
  try {
    const params = new URLSearchParams({

      limit: String(limit),
      lang: 'en',
    });
    if (typeof lat === 'number' && typeof lng === 'number') {
      params.set('lat', String(lat));
      params.set('lon', String(lng));
    }

    const res = await fetch(`${PHOTON_URL}?${params.toString()}`, {
      signal,
      headers: { Accept: 'application/json' },
    });

    if (res.ok) {
      const json = await res.json();
      const formatted = (json?.features || [])
        .map(formatPhotonPlace)
        .filter(Boolean)
        .slice(0, limit);

      if (formatted.length > 0) {
        return formatted;
      }
    }
  } catch (err) {
    if (err?.name === 'AbortError') throw err;
  }

  // 2. Fallback to Nominatim
  try {
    const params = new URLSearchParams({
      q: trimmed,
      format: 'jsonv2',
      addressdetails: '1',
      limit: String(limit),
    });
    if (typeof lat === 'number' && typeof lng === 'number') {
      const delta = 0.25;
      params.set('viewbox', `${lng - delta},${lat + delta},${lng + delta},${lat - delta}`);
      params.set('bounded', '0');
    }

    const res = await fetch(`${NOMINATIM_SEARCH_URL}?${params.toString()}`, {
      signal,
      headers: {
        Accept: 'application/json',
      },
    });

    if (!res.ok) return [];
    const items = await res.json();
    return (Array.isArray(items) ? items : [])
      .map(formatNominatimPlace)
      .filter(Boolean)
      .slice(0, limit);
  } catch (err) {
    if (err?.name === 'AbortError') throw err;
    return [];
  }
}

/**
 * Sample evenly spaced [lng, lat] points along a GeoJSON LineString coordinate list.
 */
export function sampleLineCoordinates(lngLatCoords = [], stepMeters = 150) {
  if (!Array.isArray(lngLatCoords) || lngLatCoords.length === 0) return [];
  if (lngLatCoords.length === 1) return [lngLatCoords[0]];

  const sampled = [lngLatCoords[0]];
  let accumulated = 0;

  for (let i = 1; i < lngLatCoords.length; i++) {
    const [lng1, lat1] = lngLatCoords[i - 1];
    const [lng2, lat2] = lngLatCoords[i];
    const segDist = haversineMeters(lat1, lng1, lat2, lng2);
    accumulated += segDist;

    if (accumulated >= stepMeters) {
      sampled.push([lng2, lat2]);
      accumulated = 0;
    }
  }

  const last = lngLatCoords[lngLatCoords.length - 1];
  const prev = sampled[sampled.length - 1];
  if (prev[0] !== last[0] || prev[1] !== last[1]) {
    sampled.push(last);
  }

  return sampled;
}

export function haversineMeters(lat1, lng1, lat2, lng2) {
  const R = 6371000;
  const toRad = (deg) => (deg * Math.PI) / 180;
  const dLat = toRad(lat2 - lat1);
  const dLng = toRad(lng2 - lng1);
  const a =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLng / 2) ** 2;
  return 2 * R * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
}

          ? { lat: step.maneuver.location[1], lng: step.maneuver.location[0] }
          : null,
      }))
    );

    return {
      id: `osrm-route-${idx}`,
      rawIndex: idx,
      profile: osrmProfile,
      distanceMeters,
      durationSeconds: Math.round(route.duration || 0),
      calibratedMinutes,
      calibratedDurationSeconds: calibratedMinutes * 60,
      geometryGeoJson: route.geometry,
      lngLatCoords,
      latLngCoords,
      sampledLngLat,
      steps,
    };
  });
}

/**
 * Compute bounding box around a set of [lng, lat] points with padding in degrees.
 */
function computeBBox(lngLatCoords, padDeg = 0.004) {
  let minLat = Infinity;
  let maxLat = -Infinity;
  let minLng = Infinity;
  let maxLng = -Infinity;

  for (const [lng, lat] of lngLatCoords) {
    if (lat < minLat) minLat = lat;
    if (lat > maxLat) maxLat = lat;
    if (lng < minLng) minLng = lng;
    if (lng > maxLng) maxLng = lng;
  }

  if (!Number.isFinite(minLat)) return null;

  return {
    south: Number((minLat - padDeg).toFixed(3)),
    west: Number((minLng - padDeg).toFixed(3)),
    north: Number((maxLat + padDeg).toFixed(3)),
    east: Number((maxLng + padDeg).toFixed(3)),
  };
}

    } catch (err) {
      if (err?.name === 'AbortError') throw err;
      // Try next Overpass mirror
    }
  }

  // Graceful fallback if Overpass mirrors are busy/throttled
  return {
    police: [],
    medical: [],
    activePlaces: [],
    streetLampsCount: 0,
    fromCache: false,
    degraded: true,
  };
}

  poor_lighting: -12,
  deserted: -11,
  blocked_construction: -6,

  moderate: '#d97706',
  caution: '#dc2626',

export function getTimeContext(date = new Date()) {
  const hour = date.getHours();
  const isNight = hour >= 20 || hour < 6;
  const isEveningTransition = hour >= 18 && hour < 20;

  return {
    hour,
    isNight,
    isEveningTransition,
    label: isNight
      ? 'Night Window (8 PM â€“ 6 AM)'
      : isEveningTransition
        ? 'Dusk / Evening (6 PM â€“ 8 PM)'
        : 'Daytime Window (6 AM â€“ 6 PM)',
    nightMultiplier: isNight ? 1.5 : isEveningTransition ? 1.2 : 1.0,
  };
}

export function getRecencyWeight(createdAt, nowMs = Date.now()) {
  const createdMs = createdAt ? new Date(createdAt).getTime() : nowMs;
  const ageHours = Math.max(0, (nowMs - createdMs) / (1000 * 60 * 60));

  if (ageHours <= 3) return 2.0;
  if (ageHours <= 12) return 1.6;
  if (ageHours <= 24) return 1.25;
  if (ageHours <= 48) return 0.85;
  return 0.45;
}

function countNearbyPoints(routeLatLngs, points = [], radiusMeters = 180) {
  if (!Array.isArray(points) || points.length === 0 || !Array.isArray(routeLatLngs)) {
    return 0;
  }

  const stride = Math.max(1, Math.floor(routeLatLngs.length / 80));
  let matchCount = 0;

  for (const pt of points) {
    if (typeof pt?.lat !== 'number' || typeof pt?.lng !== 'number') continue;
    let nearRoute = false;

    for (let i = 0; i < routeLatLngs.length; i += stride) {
      const [rLat, rLng] = routeLatLngs[i];
      if (haversineMeters(rLat, rLng, pt.lat, pt.lng) <= radiusMeters) {
        nearRoute = true;
        break;
      }
    }

    if (nearRoute) matchCount += 1;
  }

  return matchCount;
}

export function scoreCandidateRoute(route, context = {}) {
  const {
    reports = [],
    safetyPOIs = { police: [], medical: [], activePlaces: [], streetLampsCount: 0 },
    preferences = {
      prioritize_lighting: true,
      avoid_deserted: true,
      night_mode_strict: true,
    },
    now = new Date(),
  } = context;

  const timeCtx = getTimeContext(now);
  const nightScale =
    preferences.night_mode_strict === false
      ? Math.max(1.0, timeCtx.nightMultiplier * 0.85)
      : timeCtx.nightMultiplier;

  let score = 74;
  const reasons = [];
  const warnings = [];

  const policeNearby = countNearbyPoints(route.latLngCoords, safetyPOIs.police, 250);
  const medicalNearby = countNearbyPoints(route.latLngCoords, safetyPOIs.medical, 250);
  const activeSpotsNearby = countNearbyPoints(route.latLngCoords, safetyPOIs.activePlaces, 160);
  const lampsCount = safetyPOIs.streetLampsCount || 0;

  if (policeNearby > 0) {
    const bonus = Math.min(10, policeNearby * 5);
    score += bonus;
    reasons.push(`${policeNearby} police point${policeNearby > 1 ? 's' : ''} near corridor`);
  }

  if (medicalNearby > 0) {
    const bonus = Math.min(6, medicalNearby * 3);
    score += bonus;
    reasons.push(`${medicalNearby} clinic/hospital/pharmacy nearby`);
  }

  if (activeSpotsNearby > 0) {
    const activeBonus =
      Math.min(10, Math.round(activeSpotsNearby * 1.5)) * (timeCtx.isNight ? 1.25 : 1);
    score += activeBonus;
    reasons.push(
      `${activeSpotsNearby} open shop/cafe/fuel spot${activeSpotsNearby > 1 ? 's' : ''} along route`
    );
  } else if (timeCtx.isNight && preferences.avoid_deserted !== false) {
    score -= 6;
    warnings.push('Few active commercial spots mapped for late-night hours');
  }

  if (lampsCount > 0) {
    const lampBonus =
      Math.min(6, Math.floor(lampsCount / 4)) *
      (preferences.prioritize_lighting !== false ? 1.2 : 0.8);
    score += lampBonus;
    reasons.push(`${lampsCount} mapped streetlamp${lampsCount > 1 ? 's' : ''} in corridor area`);
  } else if (timeCtx.isNight && preferences.prioritize_lighting !== false) {
    score -= 5;
    warnings.push('Limited mapped streetlighting in this stretch');
  }

  const matchedReports = [];
  let reportDeltaTotal = 0;

  for (const rep of reports) {
    if (typeof rep?.lat !== 'number' || typeof rep?.lng !== 'number') continue;

    const samplePts = route.sampledLngLat?.length
      ? route.sampledLngLat
      : route.lngLatCoords.map(([lng, lat]) => [lng, lat]);
    let minDist = Infinity;
    for (const [lng, lat] of samplePts) {
      const d = haversineMeters(lat, lng, rep.lat, rep.lng);
      if (d < minDist) minDist = d;
    }

    if (minDist <= 160) {
      const baseWeight = CATEGORY_WEIGHTS[rep.category] ?? 0;
      const recency = getRecencyWeight(rep.created_at, now.getTime());
      const upvoteBoost = 1 + Math.min(0.6, (rep.upvotes || 0) * 0.08);

      let prefMultiplier = 1;
      if (rep.category === 'poor_lighting' && preferences.prioritize_lighting !== false) {
        prefMultiplier = 1.25;
      }
      if (rep.category === 'deserted' && preferences.avoid_deserted !== false) {
        prefMultiplier = 1.25;
      }

      const timeMultiplier =
        baseWeight < 0 && rep.category !== 'blocked_construction' ? nightScale : 1;
      const weightedDelta = baseWeight * recency * upvoteBoost * prefMultiplier * timeMultiplier;

      reportDeltaTotal += weightedDelta;
      matchedReports.push({
        ...rep,
        distanceMeters: Math.round(minDist),
        weightedDelta: Number(weightedDelta.toFixed(1)),
      });
    }
  }

  const clampedReportDelta = Math.max(-32, Math.min(24, reportDeltaTotal));
  score += clampedReportDelta;

  const positiveReports = matchedReports.filter((r) => r.weightedDelta > 0);
  const negativeReports = matchedReports.filter((r) => r.weightedDelta < 0);

  if (positiveReports.length > 0) {
    reasons.push(
      `${positiveReports.length} recent positive safety report${positiveReports.length > 1 ? 's' : ''}`
    );
  }
  for (const neg of negativeReports.slice(0, 3)) {
    const label = neg.category.replace(/_/g, ' ');
    warnings.push(`${label} reported nearby (~${neg.distanceMeters}m)`);
  }

  const finalScore = Math.max(15, Math.min(99, Math.round(score)));
  const segments = buildColoredSegments(route.latLngCoords, negativeReports, positiveReports);

  const badge =
    finalScore >= 82
      ? 'High Safety Score'
      : finalScore >= 65
        ? 'Moderate Safety'
        : 'Caution Advised';
  const badgeTone = finalScore >= 82 ? 'emerald' : finalScore >= 65 ? 'amber' : 'rose';

  if (reasons.length === 0) {
    reasons.push('Standard mapped road corridor; add live community reports to refine');
  }

  return {
    ...route,
    safetyScore: finalScore,
    badge,
    badgeTone,
    timeContext: timeCtx,
    reasons: reasons.slice(0, 4),
    warnings: warnings.slice(0, 4),
    matchedReports,
    segments,
  };
}

function buildColoredSegments(latLngCoords = [], negativeReports = [], positiveReports = []) {
  if (!Array.isArray(latLngCoords) || latLngCoords.length < 2) {
    return [];
  }

  const chunkSize = Math.max(2, Math.floor(latLngCoords.length / 10));
  const segments = [];

  for (let i = 0; i < latLngCoords.length - 1; i += chunkSize) {
    const endIdx = Math.min(latLngCoords.length - 1, i + chunkSize);
    const slice = latLngCoords.slice(i, endIdx + 1);
    const midPt = slice[Math.floor(slice.length / 2)];

    let riskScore = 0;
    for (const neg of negativeReports) {
      const dist = haversineMeters(midPt[0], midPt[1], neg.lat, neg.lng);
      if (dist <= 180) {
        riskScore += Math.abs(neg.weightedDelta || 10);
      }
    }

    let safeScore = 0;
    for (const pos of positiveReports) {
      const dist = haversineMeters(midPt[0], midPt[1], pos.lat, pos.lng);
      if (dist <= 180) {
        safeScore += pos.weightedDelta || 6;
      }
    }

    let color = SEGMENT_COLORS.safe;
    let level = 'safe';
    if (riskScore >= 16) {
      color = SEGMENT_COLORS.caution;
      level = 'caution';
    } else if (riskScore >= 7 || (safeScore === 0 && negativeReports.length > 0)) {
      color = SEGMENT_COLORS.moderate;
      level = 'moderate';
    }

    segments.push({
      positions: slice,
      color,
      level,
    });
  }

  return segments;
}

export function rankRoutesBySafety(routes = [], context = {}) {
  if (!Array.isArray(routes) || routes.length === 0) return [];

  const scored = routes.map((r) => scoreCandidateRoute(r, context));

  let safestId = scored[0].id;
  let fastestId = scored[0].id;

  for (const r of scored) {
    const currentSafest = scored.find((item) => item.id === safestId);
    if (
      r.safetyScore > currentSafest.safetyScore ||
      (r.safetyScore === currentSafest.safetyScore && r.durationSeconds < currentSafest.durationSeconds)
    ) {
      safestId = r.id;
    }

    const currentFastest = scored.find((item) => item.id === fastestId);
    if (r.durationSeconds < currentFastest.durationSeconds) {
      fastestId = r.id;
    }
  }

  return scored
    .map((r) => {
      const tags = [];
      if (r.id === safestId) tags.push('Safest Route');
      if (r.id === fastestId) tags.push('Fastest');
      if (tags.length === 0) tags.push('Alternative');

      return {
        ...r,
        routeTags: tags,
        isRecommendedSafest: r.id === safestId,
      };
    })
    .sort((a, b) => {
      if (a.isRecommendedSafest && !b.isRecommendedSafest) return -1;
      if (!a.isRecommendedSafest && b.isRecommendedSafest) return 1;
      return b.safetyScore - a.safetyScore;
    });
}

/**

 * Returns an unsubscribe cleanup function.

  if (!supabase || !tripId || typeof onUpdate !== 'function') {
    return () => {};
  }

  const channel = supabase
    .channel(`herway-trip-${tripId}`)
    .on(
      'postgres_changes',
      {
        event: 'UPDATE',
        schema: 'public',
        table: 'active_trips',
        filter: `id=eq.${tripId}`,
      },
      (payload) => {

      {activePage === 'intro' && (
        <IntroLanding
          onNavigateToMap={() => handleSelectPage('navigate')}
          onOpenSOS={() => setSosOpen(true)}
          onOpenLiveShare={() => setShareOpen(true)}
        />
      )}

      {activePage === 'navigate' && (
        <HomeNavigator
          onOpenSOS={() => setSosOpen(true)}
          onOpenLiveShare={() => setShareOpen(true)}
        />
      )}

      {activePage === 'profile' && <ProfileSafety onOpenSOS={() => setSosOpen(true)} />}

      <ShareTripModal isOpen={shareOpen} onClose={() => setShareOpen(false)} />
      <SosModal isOpen={sosOpen} onClose={() => setSosOpen(false)} />
    </div>
  );
}
